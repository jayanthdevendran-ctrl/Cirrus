[Unit]
Description=Pygame Menu
After=systemd-user-sessions.service

[Service]
Type=simple
User=cirrus
WorkingDirectory=/home/cirrus/pyscripts

# Run on the primary console
TTYPath=/dev/tty1
StandardInput=tty
StandardOutput=journal
StandardError=journal
TTYReset=yes
TTYVHangup=yes
TTYVTDisallocate=yes

Environment=SDL_VIDEODRIVER=kmsdrm

ExecStart=/usr/bin/python3 /home/cirrus/pyscripts/pygamemenu.py

Restart=always
RestartSec=2

[Install]
WantedBy=multi-user.target
